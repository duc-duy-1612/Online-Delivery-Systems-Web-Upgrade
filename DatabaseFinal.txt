-- =============================================
-- 2. TẠO BẢNG (SCHEMA) - THEO THỨ TỰ CHA TRƯỚC CON SAU
-- =============================================

-- 2.1. Tài Khoản
CREATE TABLE TaiKhoan (
    MaTK NVARCHAR(20) PRIMARY KEY,
    TenDangNhap NVARCHAR(50) NOT NULL,
    MatKhau NVARCHAR(200) NOT NULL,
    VaiTro NVARCHAR(20) NOT NULL,
    TrangThai BIT DEFAULT 1
);
GO

-- 2.2. Loại Món Ăn
CREATE TABLE LoaiMonAn (
    MaLoai NVARCHAR(20) PRIMARY KEY,
    TenLoai NVARCHAR(100),
    HinhAnh NVARCHAR(50)
);
GO

-- 2.3. Khách Hàng
CREATE TABLE KhachHang (
    MaKH NVARCHAR(20) PRIMARY KEY,
    TenKH NVARCHAR(100),
    SDT NVARCHAR(20),
    DiaChi NVARCHAR(300),
    MaTK NVARCHAR(20) FOREIGN KEY REFERENCES TaiKhoan(MaTK),
    Latitude FLOAT,
    Longitude FLOAT,
    HinhAnh NVARCHAR(100)
);
GO

-- 2.4. Shipper
CREATE TABLE Shipper (
    MaShipper NVARCHAR(20) PRIMARY KEY,
    TenShipper NVARCHAR(100),
    SDT NVARCHAR(20),
    BienSoXe NVARCHAR(50),
    MaTK NVARCHAR(20) FOREIGN KEY REFERENCES TaiKhoan(MaTK),
    HinhAnh NVARCHAR(255),
    DiemDanhGia DECIMAL(3, 1),
    ThuNhap DECIMAL(12, 2),
    Latitude FLOAT,
    Longitude FLOAT
);
GO

-- 2.5. Nhà Hàng
CREATE TABLE NhaHang (
    MaNH NVARCHAR(10) PRIMARY KEY,
    TenNH NVARCHAR(200),
    DiaChi NVARCHAR(300),
    SDT NVARCHAR(20),
    MaTK NVARCHAR(20) FOREIGN KEY REFERENCES TaiKhoan(MaTK),
    TrangThai NVARCHAR(50) DEFAULT N'Đang mở cửa',
    HinhAnh NVARCHAR(255),
    MoTa NVARCHAR(300),
    Latitude FLOAT,
    Longitude FLOAT,
    
    CONSTRAINT CHK_NhaHang_TrangThai CHECK (TrangThai IN (N'Đang mở cửa', N'Đã đóng cửa', N'Đóng cửa vĩnh viễn'))
);
GO

-- 2.6. Món Ăn
CREATE TABLE MonAn (
    MaMon NVARCHAR(20) PRIMARY KEY,
    MaNH NVARCHAR(10) FOREIGN KEY REFERENCES NhaHang(MaNH),
    TenMon NVARCHAR(200),
    MaLoai NVARCHAR(20) FOREIGN KEY REFERENCES LoaiMonAn(MaLoai),
    Gia DECIMAL(12, 2),
    MoTa NVARCHAR(500),
    HinhAnh NVARCHAR(255),
    TrangThai BIT DEFAULT 1
);
GO

-- 2.7. Đơn Hàng
CREATE TABLE DonHang (
    MaDon NVARCHAR(30) PRIMARY KEY,
    MaKH NVARCHAR(20) FOREIGN KEY REFERENCES KhachHang(MaKH),
    MaNH NVARCHAR(10) FOREIGN KEY REFERENCES NhaHang(MaNH),
    MaShipper NVARCHAR(20) FOREIGN KEY REFERENCES Shipper(MaShipper),
    TrangThai NVARCHAR(50),
    TongTien DECIMAL(12, 2),
    ThoiGianDat DATETIME DEFAULT SYSUTCDATETIME(),
    DiaChiGiaoHang NVARCHAR(300),
    SDTGiaoHang NVARCHAR(20),
    Latitude FLOAT,
    Longitude FLOAT,
    ShipperLatitude FLOAT,
    ShipperLongitude FLOAT,
    ShipFee DECIMAL(12, 2),

    CONSTRAINT CHK_DonHang_TrangThai CHECK (TrangThai IN (N'Chờ xác nhận', N'Đang lấy món', N'Đang giao', N'Hoàn thành', N'Đã hủy', N'Đã nhận đơn'))
);
GO

-- 2.8. Chi Tiết Đơn Hàng (IDENTITY PK)
CREATE TABLE ChiTietDonHang (
    MaChiTiet INT IDENTITY(1,1) PRIMARY KEY,
    MaDon NVARCHAR(30) NOT NULL FOREIGN KEY REFERENCES DonHang(MaDon),
    MaMon NVARCHAR(20) NOT NULL FOREIGN KEY REFERENCES MonAn(MaMon),
    SoLuong INT,
    DonGia DECIMAL(12, 2),
    Note NVARCHAR(MAX)
);
GO

-- 2.9. Lịch Sử Giỏ Hàng
CREATE TABLE LichSuGioHang (
    MaGH NVARCHAR(20) PRIMARY KEY,
    MaNH NVARCHAR(10) NOT NULL FOREIGN KEY REFERENCES NhaHang(MaNH),
    MaMon NVARCHAR(20) NOT NULL FOREIGN KEY REFERENCES MonAn(MaMon),
    SoLuong INT NOT NULL,
    DonGia DECIMAL(12, 2) NOT NULL,
    TongTien DECIMAL(12, 2) NOT NULL,
    ThoiGianChon DATETIME NOT NULL,
    MaKH NVARCHAR(20) NOT NULL FOREIGN KEY REFERENCES KhachHang(MaKH),
    Note NVARCHAR(MAX)
);
GO

-- 2.10. Đánh Giá (Bảng chung - nếu dùng)
CREATE TABLE DanhGia (
    MaDanhGia INT IDENTITY(1,1) PRIMARY KEY,
    MaDon NVARCHAR(30) FOREIGN KEY REFERENCES DonHang(MaDon),
    Diem INT,
    BinhLuan NVARCHAR(500)
);
GO

-- 2.11. Đánh Giá Nhà Hàng
CREATE TABLE DanhGiaNhaHang (
    MaDGNH NVARCHAR(20) PRIMARY KEY,
    MaDon NVARCHAR(30) NOT NULL FOREIGN KEY REFERENCES DonHang(MaDon),
    MaKH NVARCHAR(20) NOT NULL FOREIGN KEY REFERENCES KhachHang(MaKH),
    MaNH NVARCHAR(10) NOT NULL FOREIGN KEY REFERENCES NhaHang(MaNH),
    SoSao INT,
    BinhLuan NVARCHAR(MAX),
    ThoiGian DATETIME,
    HinhAnh NVARCHAR(255)
);
GO

-- 2.12. Đánh Giá Shipper
CREATE TABLE DanhGiaShipper (
    MaDG VARCHAR(20) PRIMARY KEY,
    MaDon NVARCHAR(30) NOT NULL FOREIGN KEY REFERENCES DonHang(MaDon),
    MaKH NVARCHAR(20) NOT NULL FOREIGN KEY REFERENCES KhachHang(MaKH),
    MaShipper NVARCHAR(20) NOT NULL FOREIGN KEY REFERENCES Shipper(MaShipper),
    SoSao INT,
    BinhLuan NVARCHAR(MAX),
    ThoiGian DATETIME
);
GO
